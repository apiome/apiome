apiome pact publish-provider-contract \
  --provider "Billing Service" \
  --provider-app-version 2.7.0+9f31c20 \
  --apiome-version billing-api/2.7.0 \
  --branch main \
  --evidence-report schemathesis.xml \
  --fail-on-unverified
# after your provider tests
schemathesis run openapi.yaml --report junit.xml

  --provider "Billing Service" \

  --evidence-report junit.xml
apiome pact publish-provider-contract openapi.yaml \
  --provider "Billing Service" \
  --provider-app-version $GIT_SHA \
  --verifier schemathesis --verifier-version 3.39.2 \
